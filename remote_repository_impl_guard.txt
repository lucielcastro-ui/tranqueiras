import 'package:dartz/dartz.dart';
import 'package:step_flutter/core/error/exceptions.dart';
import 'package:step_flutter/core/error/failures.dart';
import 'package:step_flutter/core/network/connection_checker.dart';

class RemoteRepositoryImplGuard {
  final ConnectionChecker connectionChecker;

  RemoteRepositoryImplGuard({required this.connectionChecker});

  Future<Either<Failure, T>> guard<T>(Future<T> Function() action) async {
    if (!await connectionChecker.isConnected) {
      return Left(NetworkFailure());
    }
    try {
      return Right(await action());
    } on ServerException catch (e) {
      return Left(ServerFailure(e.message));
    } catch (_) {
      return Left(ExceptionFailure());
    }
  }
}
